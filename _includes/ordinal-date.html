# FILE: _includes/ordinal-date.html # PURPOSE: Outputs a date as "DDth Month YYYY" (e.g. "22nd September 2026"), # matching the date format used in figure captions across the site # (Videos, Photos, KFI). Jekyll/Liquid has no built-in ordinal-suffix # filter, so this computes it by hand from the day-of-month. # USAGE: {% include ordinal-date.html date=some_date_value %} {%- assign day = include.date | date: "%-d" -%} {%- case day -%} {%- when '1' or '21' or '31' -%} {%- assign suffix = 'st' -%} {%- when '2' or '22' -%} {%- assign suffix = 'nd' -%} {%- when '3' or '23' -%} {%- assign suffix = 'rd' -%} {%- else -%} {%- assign suffix = 'th' -%} {%- endcase -%} {{- day -}}{{- suffix -}} {{- include.date | date: "%B %Y" -}}
